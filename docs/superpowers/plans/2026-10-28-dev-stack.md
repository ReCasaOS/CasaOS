# Local process stack Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A developer on Linux starts the real Gateway, MessageBus, UserService, LocalStorage, AppManagement and CasaOS core in a user-writable tree (`./dev/stack.sh`), then points the dashboard at that Gateway (`pnpm dev:stack`), without root and without `/etc/casaos`.

**Architecture:** CasaOS-Gateway gains `GatewayConfigFilePath` / `EnsureGatewayConfig` so a missing `gateway.ini` is created under `CASAOS_CONFIG_PATH` when set. `CasaOS/dev/writeconf.sh` writes relocated confs into `.dev-runtime/`. `CasaOS/dev/stack.sh` `go run`s sibling checkouts against that tree on `127.0.0.1:8081`. CasaOS-UI `.env.stack` + `dev/stack.mjs` wait for `stack on 127.0.0.1:8081` then serve Vue on 8080.

**Tech Stack:** Go (as in each repo's `go.mod`), bash, Node 24 / pnpm 9.0.6 for the UI glue. No new Go or npm dependencies.

func GatewayConfigFilePath() string {
	if p, ok := os.LookupEnv("CASAOS_CONFIG_PATH"); ok && p != "" {
		return filepath.Join(p, GatewayName+"."+GatewayConfigType)
	}
	return filepath.Join(constants.DefaultConfigPath, GatewayName+"."+GatewayConfigType)
}

func EnsureGatewayConfig(sample string) error {
	path := GatewayConfigFilePath()
	_, err := os.Stat(path)
	if err == nil {
		return nil
	}
	if !os.IsNotExist(err) {
		return err
	}
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		return err
	}
	return os.WriteFile(path, []byte(sample), 0o600)
}

```bash
#!/usr/bin/env bash
set -euo pipefail

if [[ "$(uname -s)" != "Linux" ]]; then
	echo "the local stack is Linux only" >&2
	exit 1
fi
if [[ "$(id -u)" -eq 0 ]]; then
	echo "don't run the stack as root" >&2
	exit 1
fi

HERE=$(cd "$(dirname "$0")" && pwd)
CASAOS_REPO=$(cd "$HERE/.." && pwd)
PARENT=$(cd "$CASAOS_REPO/.." && pwd)
DEVROOT="$CASAOS_REPO/.dev-runtime"
WRITECONF="$HERE/writeconf.sh"

need() {
	if [[ ! -d "$PARENT/$1" ]]; then
		echo "missing $PARENT/$1" >&2
		exit 1
	fi
}
need CasaOS-Gateway
need CasaOS-MessageBus
need CasaOS-UserService

http_code() {
	curl -sS -o /dev/null -w '%{http_code}' --max-time 1 "$1" || echo 000
}

if [[ "$(http_code http://127.0.0.1:8081/ping)" == "200" ]]; then
	echo "127.0.0.1:8081 is already in use" >&2
	exit 1
fi

mkdir -p "$DEVROOT/run"
rm -f "$DEVROOT/run/"*.url "$DEVROOT/run/"*.pid "$DEVROOT/run/internal.secret"

bash "$WRITECONF" "$DEVROOT" "$CASAOS_REPO"

PIDS=()
REQUIRED=()

stop() {
	local code=$1
	trap - INT TERM
	local pid
	for pid in "${PIDS[@]+"${PIDS[@]}"}"; do
		kill -TERM -- "-$pid" 2>/dev/null || kill -TERM "$pid" 2>/dev/null || true
	done
	for pid in "${PIDS[@]+"${PIDS[@]}"}"; do
		wait "$pid" 2>/dev/null || true
	done
	exit "$code"
}
trap 'stop 0' INT TERM

spawn() {
	local cwd=$1
	shift
	(
		cd "$cwd"
		export CASAOS_CONFIG_PATH="$DEVROOT/etc"
		exec setsid go run "$@"
	) &
	PIDS+=("$!")
}

alive() {
	kill -0 "$1" 2>/dev/null
}

wait_file() {
	local path=$1 pid=$2 label=$3
	local deadline=$((SECONDS + 120))
	while (( SECONDS < deadline )); do
		if ! alive "$pid"; then
			echo "$label exited" >&2
			stop 1
		fi
		if [[ -f "$path" ]]; then
			return 0
		fi
		sleep 0.5
	done
	if [[ "$label" == gateway ]]; then
		echo "gateway did not answer on 127.0.0.1:8081" >&2
	else
		echo "$label did not become ready" >&2
	fi
	stop 1
}

wait_ping() {
	local pid=$1
	local deadline=$((SECONDS + 120))
	while (( SECONDS < deadline )); do
		if ! alive "$pid"; then
			echo "gateway did not answer on 127.0.0.1:8081" >&2
			stop 1
		fi
		if [[ "$(http_code http://127.0.0.1:8081/ping)" == "200" ]]; then
			return 0
		fi
		sleep 0.5
	done
	echo "gateway did not answer on 127.0.0.1:8081" >&2
	stop 1
}

spawn "$DEVROOT/etc" "$PARENT/CasaOS-Gateway" -w "$DEVROOT/www"
REQUIRED+=("${PIDS[-1]}")
wait_ping "${PIDS[-1]}"
wait_file "$DEVROOT/run/management.url" "${PIDS[-1]}" gateway

spawn "$PARENT/CasaOS-MessageBus" . -c "$DEVROOT/etc/message-bus.conf"
REQUIRED+=("${PIDS[-1]}")
wait_file "$DEVROOT/run/message-bus.url" "${PIDS[-1]}" message-bus

spawn "$PARENT/CasaOS-UserService" . -c "$DEVROOT/etc/user-service.conf"
REQUIRED+=("${PIDS[-1]}")
wait_file "$DEVROOT/run/user-service.url" "${PIDS[-1]}" user-service

if [[ -d "$PARENT/CasaOS-LocalStorage" ]]; then
	spawn "$PARENT/CasaOS-LocalStorage" . -c "$DEVROOT/etc/local-storage.conf"
	ls_pid="${PIDS[-1]}"
	ls_ok=0
	deadline=$((SECONDS + 120))
	while (( SECONDS < deadline )); do
		if ! alive "$ls_pid"; then
			echo "warning: LocalStorage exited; skipping" >&2
			break
		fi
		code=$(http_code http://127.0.0.1:8081/v1/disks)
		if [[ "$code" != "404" && "$code" != "000" ]]; then
			ls_ok=1
			break
		fi
		sleep 0.5
	done
	if [[ "$ls_ok" -eq 0 ]] && alive "$ls_pid"; then
		echo "warning: LocalStorage did not register /v1/disks; continuing" >&2
	fi
else
	echo "warning: missing $PARENT/CasaOS-LocalStorage; skipping" >&2
fi

if [[ -d "$PARENT/CasaOS-AppManagement" ]]; then
	if [[ ! -S /var/run/docker.sock && -z "${DOCKER_HOST:-}" ]]; then
		echo "warning: no Docker socket; compose calls will fail" >&2
	fi
	spawn "$PARENT/CasaOS-AppManagement" . -c "$DEVROOT/etc/app-management.conf"
	REQUIRED+=("${PIDS[-1]}")
	wait_file "$DEVROOT/run/app-management.url" "${PIDS[-1]}" app-management
else
	echo "warning: missing $PARENT/CasaOS-AppManagement; skipping" >&2
fi

spawn "$CASAOS_REPO" . -c "$DEVROOT/etc/casaos.conf"
REQUIRED+=("${PIDS[-1]}")
wait_file "$DEVROOT/run/casaos.url" "${PIDS[-1]}" casaos

echo "stack on 127.0.0.1:8081"

while true; do
	for pid in "${REQUIRED[@]}"; do
		if ! alive "$pid"; then
			echo "a required process exited" >&2
			stop 1
		fi
	done
	sleep 1
done
```

Notes for the implementer:

- `${PIDS[-1]}` needs bash 4+. Linux bash is 4+.
- `${PIDS[@]+"${PIDS[@]}"}` avoids `unbound variable` under `set -u` if `stop` runs before any spawn.
- Ping timeout and gateway url-file timeout both print `gateway did not answer on 127.0.0.1:8081`. Other services print `$label did not become ready`. A stack already bound to 8081 before we spawn prints `127.0.0.1:8081 is already in use`.
- AppManagement is optional in the spec when the checkout is missing; when the checkout exists it is started and its url file is required (failure stops the stack). That is what the script above does.
- Do not inherit a parent `CASAOS_CONFIG_PATH`: `export CASAOS_CONFIG_PATH="$DEVROOT/etc"` in the spawn subshell overwrites it for the child.

import { spawn } from 'node:child_process'
import { existsSync } from 'node:fs'
import process from 'node:process'
import { fileURLToPath } from 'node:url'

const root = fileURLToPath(new URL('..', import.meta.url))
const casaos = fileURLToPath(new URL('../../CasaOS', import.meta.url))
const stackSh = fileURLToPath(new URL('../../CasaOS/dev/stack.sh', import.meta.url))
const WAIT_MS = 300000
const READY = 'stack on 127.0.0.1:8081'

if (!existsSync(stackSh)) {
	console.error('clone CasaOS next to this repository and run ./dev/stack.sh from there, or keep this checkout beside it')
	process.exit(1)
}

function waitForStack(child, deadline) {
	return new Promise((resolve, reject) => {
		const fail = () => {
			reject(new Error('stack did not become ready on 127.0.0.1:8081'))
		}
		if (child.exitCode !== null || child.signalCode !== null) {
			fail()
			return
		}
		const timer = setTimeout(() => {
			child.off('exit', onExit)
			fail()
		}, Math.max(0, deadline - Date.now()))
		function onExit() {
			clearTimeout(timer)
			fail()
		}
		child.once('exit', onExit)
		let buf = ''
		child.stdout.on('data', (chunk) => {
			process.stdout.write(chunk)
			buf += chunk
			if (buf.includes(READY)) {
				clearTimeout(timer)
				child.off('exit', onExit)
				resolve()
			}
		})
	})
}

function kill(child) {
	if (!child || child.exitCode !== null || child.signalCode !== null) {
		return
	}
	try {
		process.kill(-child.pid, 'SIGTERM')
	} catch {
		child.kill('SIGTERM')
	}
}

const stack = spawn('bash', [stackSh], {
	cwd: casaos,
	stdio: ['ignore', 'pipe', 'inherit'],
	detached: true,
})

try {
	await waitForStack(stack, Date.now() + WAIT_MS)
} catch (error) {
	kill(stack)
	console.error(error.message)
	process.exit(1)
}

const vue = spawn('pnpm', ['exec', 'vue-cli-service', 'serve', '--mode', 'stack'], {
	cwd: root,
	stdio: 'inherit',
	detached: true,
})

let stopping = false
function stop(code) {
	if (stopping) {
		return
	}
	stopping = true
	kill(vue)
	kill(stack)
	process.exit(code)
}

process.on('SIGINT', () => stop(0))
process.on('SIGTERM', () => stop(0))
vue.on('exit', (code, signal) => {
	if (stopping) {
		return
	}
	kill(stack)
	process.exit(code ?? (signal ? 1 : 0))
})
stack.on('exit', () => {
	if (stopping) {
		return
	}
	kill(vue)
	process.exit(1)
})
